def make_light_llm_client() -> LLMClient:
    """light-model factory. 读 LLM_LIGHT_* env, 缺则 fallback 主 LLM_*.

    Phase 17.2 Feature B: cheap LLM (haiku / gpt-4o-mini) 给 classify /
    canonical signature 这种简单 task 用. 失败由 with_light_fallback
    自动回退到主 LLM.
    """
    load_dotenv(override=False)
    proto = os.environ.get("LLM_LIGHT_PROTOCOL") or os.environ["LLM_PROTOCOL"]
    base_url = os.environ.get("LLM_LIGHT_BASE_URL") or os.environ["LLM_BASE_URL"]
    api_key = os.environ.get("LLM_LIGHT_API_KEY") or os.environ["LLM_API_KEY"]
    model = os.environ.get("LLM_LIGHT_MODEL") or os.environ["LLM_MODEL"]
    max_tokens_env = os.environ.get("LLM_LIGHT_MAX_TOKENS") or os.environ.get(
        "LLM_MAX_TOKENS"
    )

    max_tokens: int | None = None
    if max_tokens_env is not None:
        max_tokens = int(max_tokens_env)
        if max_tokens < 1:
            raise ValueError(f"LLM_LIGHT_MAX_TOKENS must be >= 1, got {max_tokens}")

    if proto == "anthropic":
        return AnthropicProtocolClient(
            api_key=api_key,
            default_model=model,
            base_url=base_url,
            max_tokens=max_tokens,
        )
    if proto == "openai":
        mode_str = os.environ.get("LLM_STRUCTURED_OUTPUT_MODE", "json_schema")
        mode: Mode = mode_str  # type: ignore[assignment]
        return OpenAIProtocolClient(
            api_key=api_key,
            default_model=model,
            base_url=base_url,
            mode=mode,
            max_tokens=max_tokens,
        )
    raise ValueError(
        f"Unknown LLM_LIGHT_PROTOCOL (or LLM_PROTOCOL): {proto!r}, "
        f"must be 'anthropic' or 'openai'"
    )
```

# src/explain_engine/llm/light_fallback.py
"""light_llm + 主 LLM auto-fallback helper — Phase 17.2 Feature B.

light 失败 (LLMError / SchemaValidationError) → 自动 fallback 主 LLM,
log warning. Caller 零感知.
"""
import logging
from collections.abc import Awaitable, Callable
from typing import TypeVar

from explain_engine.llm.client import LLMClient
from explain_engine.llm.errors import LLMError, SchemaValidationError

logger = logging.getLogger(__name__)

T = TypeVar("T")


async def with_light_fallback(
    light_llm: LLMClient,
    main_llm: LLMClient,
    fn: Callable[[LLMClient], Awaitable[T]],
) -> T:
    """跑 fn(light_llm), 失败 (LLM / Schema 错) 自动 fallback fn(main_llm).

    若 light is main (same client instance), 直接 fn(main_llm) — 无 fallback.
    """
    if light_llm is main_llm:
        return await fn(main_llm)

    try:
        return await fn(light_llm)
    except (LLMError, SchemaValidationError) as exc:
        logger.warning(
            "light_llm 失败, fallback 主 LLM: %s",
            exc,
        )
        return await fn(main_llm)
```

from explain_engine.llm.light_fallback import with_light_fallback

QuestionType = Literal["causal_modern", "concept_explanation", "mechanism", "phenomenon"]
_VALID_TYPES = {"causal_modern", "concept_explanation", "mechanism", "phenomenon"}


class _ClassifyOutput(BaseModel):
    type: str


async def _classify_question(
    question: str,
    light_llm: LLMClient,
    main_llm: LLMClient,
) -> QuestionType:
    """前置 classify, 失败/未知 type fallback 'causal_modern'."""
    prompt = load_prompt("question_classify")
    messages = [
        Message(role="system", content=prompt["system"]),
        Message(role="user", content=prompt["user_template"].format(question=question)),
    ]

    async def _do_call(llm: LLMClient):
        return await llm.chat(messages, schema=_ClassifyOutput)

    try:
        resp = await with_light_fallback(light_llm, main_llm, _do_call)
    except Exception as exc:
        logger.warning("classify 整体失败, fallback causal_modern: %s", exc)
        return "causal_modern"

    if resp.parsed is None:
        return "causal_modern"
    type_str = resp.parsed.get("type") if isinstance(resp.parsed, dict) else getattr(resp.parsed, "type", None)
    if type_str not in _VALID_TYPES:
        logger.warning("classify 返未知 type %r, fallback causal_modern", type_str)
        return "causal_modern"
    return type_str  # type: ignore[return-value]

async def _handle_delete(chat: "ChatSession", args: list[str]) -> "list[ChatEvent]":
    from explain_engine.chat.chat_copy import (
        err_delete_active, err_delete_not_found, msg_delete_done,
    )
    from explain_engine.chat.session import SlashEvent
    from explain_engine.persistence.session import SessionStore

    force = "--force" in args
    pos_args = [a for a in args if a != "--force"]
    if not pos_args:
        return [SlashEvent(type="slash_error", content="用法: /delete <sid> [--force]")]
    sid = pos_args[0]

    if chat.session.meta.session_id == sid:
        return [SlashEvent(type="slash_error", content=err_delete_active(sid))]

    if not force:
        # slash REPL 内部 input() 走 prompt_toolkit (看现 chat REPL 怎么读 stdin)
        # 简单方案: 直接拒绝, 提示 --force.
        return [SlashEvent(type="slash_error",
                           content="slash /delete 默认拒绝, 加 --force 跳过二次确认")]
        # 或更 honest: 实现 confirm prompt. 本 plan 推荐简单方案防 chat I/O 复杂.

    store = SessionStore()
    try:
        store.delete(sid)
    except FileNotFoundError:
        return [SlashEvent(type="slash_error", content=err_delete_not_found(sid))]

    return [SlashEvent(type="slash_delete", content=msg_delete_done(sid))]
